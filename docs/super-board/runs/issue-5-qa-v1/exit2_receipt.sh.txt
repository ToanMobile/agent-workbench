#!/usr/bin/env bash
# QA probe issue #5: a linked worktree whose receipt for the pushed content is NOT an exit 0 (an UNVERIFIED exit 2 the gate keeps
# with the real fingerprint and head) must not cover a push from main. usage: exit2_receipt.sh <devkit dir>
. "$1/tests/lib/clean_git_env.sh"
set -u
DEVKIT_DIR="$(cd "$1" && pwd)"
GATE="$DEVKIT_DIR/bin/post-fix-gate.py"; PG="$DEVKIT_DIR/bin/push_gate.py"
TMP="$(mktemp -d)"; trap 'rm -rf "$TMP"' EXIT
R="$TMP/repo"; W="$TMP/wt"
mkdir -p "$R/src" "$R/.agents" && cd "$R" || exit 1
git init -q -b main . && git config user.email t@t && git config user.name t && git config commit.gpgsign false
echo "fun ok() = 1" > src/Core.kt
cat > .agents/regression_matrix.active.json <<'JSON'
{"project":"t","rules":[{"component":"Core","watch_files":["src/*"],
 "mandatory_regression_tests":[{"id":"REG-1","name":"core","command":"true"}]}]}
JSON
printf '.agents/CHECKLIST.md\n.agents/regression_status.json\n.agents/evidence/\n.agents/regression_journal/\n.claude/\n' > .gitignore
git add -A && git commit -qm init
git worktree add -q -b task "$W" main
echo "fun ok() = 2" > "$W/src/Core.kt"
CLAUDE_PROJECT_DIR="$W" python3 "$GATE" --run-tests --full --brief > "$TMP/gate.out" 2>&1; echo "gate exit=$?"
git -C "$W" commit -qam "task work" && git -C "$R" merge -q --ff-only task; echo "commit+merge exit=$?"
RW="$(git -C "$W" rev-parse --absolute-git-dir)/postfix-gate/full_pass.json"
python3 "$PG" "$R" >/dev/null; echo "control (exit-0 receipt) push_gate exit=$?"
python3 - "$RW" <<'PY'
import json, sys
r = json.load(open(sys.argv[1])); r["exit"] = 2; json.dump(r, open(sys.argv[1], "w"))
PY
OUT="$(python3 "$PG" "$R" 2>&1)"; echo "exit-2 receipt push_gate exit=$? :: ${OUT:0:200}"
